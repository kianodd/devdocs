One shape parameter [c>0] and support [xinleft[0,1/cright]].

begin{eqnarray*} fleft(x;cright) & = & frac{2left(1-cxright)^{frac{1}{c}-1}}{left(1+left(1-cxright)^{1/c}right)^{2}}\ Fleft(x;cright) & = & frac{1-left(1-cxright)^{1/c}}{1+left(1-cxright)^{1/c}}\ Gleft(q;cright) & = & frac{1}{c}left[1-left(frac{1-q}{1+q}right)^{c}right]end{eqnarray*}

begin{eqnarray*} hleft[Xright] & = & 2-left(2c+1right)log2.end{eqnarray*}
